from typing import Any
import numpy as np
from maniskill_backend.dynamic_adapter import ManiSkillDynamicRobot


class GeneratedSelectiveRobot(ManiSkillDynamicRobot):
    def __init__(self, env: Any, *, control_mode: str, robot_uid: str) -> None:
        super().__init__(env, control_mode=control_mode, robot_uid=robot_uid,
                         gripper_open=0.0, gripper_close=0.81,
                         move_steps=30, settle_steps=12, max_delta_m=0.04)

    def _stage_approach(self) -> bool:
        cube_pos = np.asarray(self._entity_pos("cube"), dtype=np.float32).reshape(-1)[:3]
        tcp_pos = np.asarray(self._tcp_pos(), dtype=np.float32).reshape(-1)[:3]
        pregrasp = cube_pos + np.array([0.0, 0.0, 0.10], dtype=np.float32)
        for _ in range(6):
            if self._early_stop():
                return False
            self._move_towards(pregrasp, gripper=self.gripper_open, steps=self.move_steps)
            tcp_pos = np.asarray(self._tcp_pos(), dtype=np.float32).reshape(-1)[:3]
            if float(np.linalg.norm(tcp_pos - pregrasp)) <= 0.02:
                break
        tcp_pos = np.asarray(self._tcp_pos(), dtype=np.float32).reshape(-1)[:3]
        cube_pos = np.asarray(self._entity_pos("cube"), dtype=np.float32).reshape(-1)[:3]
        above = tcp_pos[2] > cube_pos[2] + 0.05
        lateral = float(np.linalg.norm(tcp_pos[:2] - cube_pos[:2])) <= 0.05
        return bool(above and lateral)

    def _stage_grasp(self) -> bool:
        cube_pos = np.asarray(self._entity_pos("cube"), dtype=np.float32).reshape(-1)[:3]
        grasp_target = cube_pos + np.array([0.0, 0.0, 0.005], dtype=np.float32)
        for _ in range(6):
            if self._early_stop():
                return False
            self._move_towards(grasp_target, gripper=self.gripper_open, steps=self.move_steps)
            tcp_pos = np.asarray(self._tcp_pos(), dtype=np.float32).reshape(-1)[:3]
            if float(np.linalg.norm(tcp_pos - grasp_target)) <= 0.015:
                break
        if self._early_stop():
            return False
        self._repeat_action(np.zeros(3, dtype=np.float32),
                            gripper=self.gripper_close, steps=self.settle_steps)
        for _ in range(4):
            if self._early_stop():
                return False
            if self._is_grasping_entity("cube"):
                return True
            self._repeat_action(np.zeros(3, dtype=np.float32),
                                gripper=self.gripper_close, steps=self.settle_steps)
        return bool(self._is_grasping_entity("cube"))

    def _stage_align(self) -> bool:
        goal_pos = np.asarray(self._entity_pos("goal_site"), dtype=np.float32).reshape(-1)[:3]
        lift_target = goal_pos + np.array([0.0, 0.0, 0.06], dtype=np.float32)
        for _ in range(8):
            if self._early_stop():
                return False
            self._move_towards(lift_target, gripper=self.gripper_close, steps=self.move_steps)
            tcp_pos = np.asarray(self._tcp_pos(), dtype=np.float32).reshape(-1)[:3]
            if float(np.linalg.norm(tcp_pos - lift_target)) <= 0.02:
                break
        for _ in range(8):
            if self._early_stop():
                return False
            self._move_towards(goal_pos, gripper=self.gripper_close, steps=self.move_steps)
            cube_pos = np.asarray(self._entity_pos("cube"), dtype=np.float32).reshape(-1)[:3]
            goal_pos = np.asarray(self._entity_pos("goal_site"), dtype=np.float32).reshape(-1)[:3]
            if float(np.linalg.norm(cube_pos - goal_pos)) <= 0.02:
                break
        if self._early_stop():
            return False
        self._repeat_action(np.zeros(3, dtype=np.float32),
                            gripper=self.gripper_close, steps=self.settle_steps)
        eval_result = self._official_evaluation()
        return bool(eval_result.get("is_obj_placed", False))

    def run_source_actions(self) -> bool:
        if not self._run_stage("approach", self._stage_approach): return False
        if not self._run_stage("grasp", self._stage_grasp): return False
        if not self._run_stage("align", self._stage_align): return False
        return self._official_success()


def build_robot(env: Any, *, control_mode: str, robot_uid: str) -> GeneratedSelectiveRobot:
    return GeneratedSelectiveRobot(env, control_mode=control_mode, robot_uid=robot_uid)
