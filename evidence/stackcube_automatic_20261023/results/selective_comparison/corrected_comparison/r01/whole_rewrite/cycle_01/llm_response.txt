from typing import Any
import numpy as np
from maniskill_backend.dynamic_adapter import ManiSkillDynamicRobot


class GeneratedRobot(ManiSkillDynamicRobot):
    def __init__(self, env: Any, *, control_mode: str, robot_uid: str) -> None:
        super().__init__(env, control_mode=control_mode, robot_uid=robot_uid,
                         gripper_open=0.0, gripper_close=0.81,
                         move_steps=30, settle_steps=12, max_delta_m=0.04)

    # ---- helpers -------------------------------------------------------
    def _goto(self, target, gripper, steps=None, tol=0.008, max_iters=40):
        target = np.asarray(target, dtype=np.float64).reshape(3)
        steps = steps if steps is not None else self.move_steps
        for _ in range(max_iters):
            if self._early_stop():
                return False
            cur = np.asarray(self._tcp_pos(), dtype=np.float64).reshape(3)
            if np.linalg.norm(cur - target) <= tol:
                return True
            self._move_towards(target, gripper=gripper, steps=steps)
        return True

    def _hold(self, gripper, steps):
        for _ in range(steps):
            if self._early_stop():
                return False
            self._repeat_action(np.zeros(3), gripper=gripper, steps=1)
        return True

    def _grasp_pose_for_cube(self, cube_pos):
        # Top-down grasp: TCP directly above cube center, small downward offset.
        cube_pos = np.asarray(cube_pos, dtype=np.float64).reshape(3)
        grasp = cube_pos.copy()
        grasp[2] = cube_pos[2] + 0.005
        return grasp

    # ---- main policy ---------------------------------------------------
    def run_source_actions(self) -> bool:
        # Stage 1: approach — move above cubeA
        cubeA = np.asarray(self._entity_pos("cubeA"), dtype=np.float64).reshape(3)
        cubeB = np.asarray(self._entity_pos("cubeB"), dtype=np.float64).reshape(3)

        pregrasp = cubeA.copy()
        pregrasp[2] = cubeA[2] + 0.12
        if not self._goto(pregrasp, gripper=self.gripper_open):
            return False

        # Stage 2: grasp — descend to grasp pose and close gripper
        grasp_pose = self._grasp_pose_for_cube(cubeA)
        if not self._goto(grasp_pose, gripper=self.gripper_open):
            return False
        if not self._hold(self.gripper_close, self.settle_steps):
            return False
        if not self._hold(self.gripper_close, self.settle_steps):
            return False

        # Stage 3: lift — raise cubeA while keeping gripper closed
        lift_target = grasp_pose.copy()
        lift_target[2] = cubeA[2] + 0.12
        if not self._goto(lift_target, gripper=self.gripper_close):
            return False

        # Re-measure to compute stack alignment relative to cubeB
        cubeA_now = np.asarray(self._entity_pos("cubeA"), dtype=np.float64).reshape(3)
        cubeB_now = np.asarray(self._entity_pos("cubeB"), dtype=np.float64).reshape(3)
        tcp_now = np.asarray(self._tcp_pos(), dtype=np.float64).reshape(3)

        # Stage 4: align — move above cubeB at stack height
        stack_z = cubeB_now[2] + 0.04 + 0.005  # cube half size * 2 + small clearance
        align_target = np.array([cubeB_now[0], cubeB_now[1], stack_z], dtype=np.float64)
        # Preserve TCP-to-held-object offset in xy
        offset_xy = tcp_now[:2] - cubeA_now[:2]
        align_target[:2] = align_target[:2] + offset_xy
        if not self._goto(align_target, gripper=self.gripper_close):
            return False

        # Stage 5: release — open gripper and wait for static success
        if not self._hold(self.gripper_open, self.settle_steps):
            return False
        if not self._hold(self.gripper_open, self.settle_steps):
            return False

        # Allow physics to settle; check official success in loop
        for _ in range(60):
            if self._early_stop():
                break
            ev = self._official_evaluation()
            if bool(ev.get("success", False)):
                return True
            self._repeat_action(np.zeros(3), gripper=self.gripper_open, steps=2)

        ev = self._official_evaluation()
        return bool(ev.get("success", False))


def build_robot(env: Any, *, control_mode: str, robot_uid: str) -> GeneratedRobot:
    return GeneratedRobot(env, control_mode=control_mode, robot_uid=robot_uid)
