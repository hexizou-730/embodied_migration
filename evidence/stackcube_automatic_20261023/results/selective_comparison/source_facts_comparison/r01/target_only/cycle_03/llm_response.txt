from typing import Any
import numpy as np
from maniskill_backend.dynamic_adapter import ManiSkillDynamicRobot


class GeneratedRobot(ManiSkillDynamicRobot):
    def __init__(self, env: Any, *, control_mode: str, robot_uid: str) -> None:
        super().__init__(env, control_mode=control_mode, robot_uid=robot_uid,
                         gripper_open=0.0, gripper_close=0.81,
                         move_steps=30, settle_steps=12, max_delta_m=0.04)

    # ---------- safe wrappers ----------
    def _safe_move(self, target, gripper, steps):
        try:
            self._move_towards(np.asarray(target, dtype=np.float64),
                               gripper=float(gripper), steps=int(steps))
        except Exception:
            pass

    def _safe_repeat(self, delta, gripper, steps):
        try:
            self._repeat_action(np.asarray(delta, dtype=np.float64),
                                gripper=float(gripper), steps=int(steps))
        except Exception:
            pass

    def _stop(self):
        try:
            return bool(self._early_stop())
        except Exception:
            return False

    def _pos(self, name):
        try:
            return np.asarray(self._entity_pos(name), dtype=np.float64).reshape(-1)[:3]
        except Exception:
            return None

    def _tcp(self):
        try:
            return np.asarray(self._tcp_pos(), dtype=np.float64).reshape(-1)[:3]
        except Exception:
            return None

    def _grasped(self, name):
        try:
            return bool(self._is_grasping_entity(name))
        except Exception:
            return False

    def _success(self):
        try:
            return bool(self._official_evaluation().get("success", False))
        except Exception:
            return False

    # ---------- motion primitives ----------
    def _move_xy(self, target_xy, gripper, steps, tol=0.005, max_iters=12):
        """Iteratively drive TCP xy toward target using measured feedback."""
        for _ in range(max_iters):
            if self._stop():
                return True
            tcp = self._tcp()
            if tcp is None:
                return False
            err = np.asarray(target_xy, dtype=np.float64) - tcp[:2]
            if float(np.linalg.norm(err)) <= tol:
                return False
            step = np.clip(err, -0.025, 0.025)
            goal = np.array([tcp[0] + step[0], tcp[1] + step[1], tcp[2]])
            self._safe_move(goal, gripper, steps)
        return False

    def _move_z(self, target_z, gripper, steps, tol=0.004, max_iters=12):
        for _ in range(max_iters):
            if self._stop():
                return True
            tcp = self._tcp()
            if tcp is None:
                return False
            dz = float(target_z) - float(tcp[2])
            if abs(dz) <= tol:
                return False
            step = float(np.clip(dz, -0.025, 0.025))
            goal = np.array([tcp[0], tcp[1], tcp[2] + step])
            self._safe_move(goal, gripper, steps)
        return False

    def _move_xyz(self, target, gripper, steps, tol=0.006, max_iters=14):
        for _ in range(max_iters):
            if self._stop():
                return True
            tcp = self._tcp()
            if tcp is None:
                return False
            err = np.asarray(target, dtype=np.float64) - tcp
            if float(np.linalg.norm(err)) <= tol:
                return False
            step = np.clip(err, -0.025, 0.025)
            goal = tcp + step
            self._safe_move(goal, gripper, steps)
        return False

    def _open_gripper(self, steps=20):
        self._safe_repeat(np.zeros(3), 0.0, steps)

    def _close_gripper(self, steps=25):
        self._safe_repeat(np.zeros(3), 0.81, steps)

    # ---------- grasp attempt ----------
    def _attempt_grasp(self, obj_name, grasp_z, hover_z):
        """Approach from above, descend, close, verify grasp."""
        pos = self._pos(obj_name)
        if pos is None:
            return False

        # Hover above object
        hover = np.array([pos[0], pos[1], hover_z])
        self._move_xyz(hover, 0.0, 40)
        if self._stop():
            return False

        # Fine xy alignment at hover height
        self._move_xy(pos[:2], 0.0, 20)
        if self._stop():
            return False

        # Descend to grasp height
        self._move_z(grasp_z, 0.0, 25)
        if self._stop():
            return False

        # Re-align xy at grasp height (small correction)
        pos = self._pos(obj_name)
        if pos is not None:
            self._move_xy(pos[:2], 0.0, 15)
        if self._stop():
            return False

        # Close gripper
        self._close_gripper(25)
        if self._stop():
            return False

        # Verify grasp
        if self._grasped(obj_name):
            return True

        # Retry: descend a bit more and close again
        tcp = self._tcp()
        if tcp is not None:
            goal = np.array([tcp[0], tcp[1], tcp[2] - 0.012])
            self._safe_move(goal, 0.0, 15)
            self._close_gripper(20)
            if self._stop():
                return False
            if self._grasped(obj_name):
                return True

        # Second retry: open, re-align, descend deeper, close
        self._open_gripper(15)
        pos = self._pos(obj_name)
        if pos is not None:
            self._move_xy(pos[:2], 0.0, 15)
            self._move_z(grasp_z - 0.008, 0.0, 20)
        self._close_gripper(25)
        if self._stop():
            return False
        return self._grasped(obj_name)

    def _lift_and_verify(self, obj_name, lift_z):
        """Lift and verify object moved with gripper."""
        pos_before = self._pos(obj_name)
        tcp = self._tcp()
        if tcp is None:
            return False
        lift = np.array([tcp[0], tcp[1], lift_z])
        self._move_xyz(lift, 0.81, 40)
        if self._stop():
            return False
        pos_after = self._pos(obj_name)
        if pos_before is None or pos_after is None:
            return self._grasped(obj_name)
        moved = float(np.linalg.norm(pos_after - pos_before))
        return moved > 0.02 or self._grasped(obj_name)

    # ---------- main policy ----------
    def run_source_actions(self) -> bool:
        posA = self._pos("cubeA")
        posB = self._pos("cubeB")
        if posA is None or posB is None:
            return False

        cube_half = 0.02
        # Grasp near top of cubeA (slightly above center to avoid pushing)
        grasp_z_A = float(posA[2]) + cube_half * 0.6
        hover_z_A = float(posA[2]) + 0.14
        lift_z_A = float(posA[2]) + 0.18

        # Phase 1: grasp cubeA
        grasped = self._attempt_grasp("cubeA", grasp_z_A, hover_z_A)
        if self._stop():
            return self._success()

        if not grasped:
            # Retry with lower grasp height
            posA = self._pos("cubeA")
            if posA is None:
                return False
            grasp_z_A = float(posA[2]) + cube_half * 0.2
            hover_z_A = float(posA[2]) + 0.14
            lift_z_A = float(posA[2]) + 0.18
            grasped = self._attempt_grasp("cubeA", grasp_z_A, hover_z_A)
            if self._stop():
                return self._success()

        if not grasped:
            return self._success()

        # Lift and verify object follows
        lifted = self._lift_and_verify("cubeA", lift_z_A)
        if self._stop():
            return self._success()

        if not lifted:
            # Re-grasp attempt from current position
            posA = self._pos("cubeA")
            if posA is None:
                return False
            grasp_z_A = float(posA[2]) + cube_half * 0.4
            hover_z_A = float(posA[2]) + 0.14
            lift_z_A = float(posA[2]) + 0.18
            grasped = self._attempt_grasp("cubeA", grasp_z_A, hover_z_A)
            if self._stop():
                return self._success()
            if not grasped:
                return self._success()
            lifted = self._lift_and_verify("cubeA", lift_z_A)
            if self._stop():
                return self._success()
            if not lifted:
                return self._success()

        # Phase 2: transport above cubeB
        posB = self._pos("cubeB")
        if posB is None:
            return self._success()

        above_B = np.array([posB[0], posB[1], float(posB[2]) + 0.20])
        self._move_xyz(above_B, 0.81, 60)
        if self._stop():
            return self._success()

        # Align xy over cubeB
        posB = self._pos("cubeB")
        if posB is not None:
            self._move_xy(posB[:2], 0.81, 25)
        if self._stop():
            return self._success()

        # Phase 3: lower to stack height
        posB = self._pos("cubeB")
        if posB is None:
            return self._success()
        stack_z = float(posB[2]) + cube_half * 2 + 0.006
        self._move_z(stack_z, 0.81, 25)
        if self._stop():
            return self._success()

        # Fine xy correction at stack height
        posB = self._pos("cubeB")
        if posB is not None:
            self._move_xy(posB[:2], 0.81, 15)
        if self._stop():
            return self._success()

        # Release
        self._open_gripper(25)
        if self._stop():
            return self._success()

        # Retreat upward to let cube settle
        retreat = np.array([posB[0], posB[1], stack_z + 0.14])
        self._move_xyz(retreat, 0.0, 40)
        if self._stop():
            return self._success()

        # Settle in place
        self._safe_repeat(np.zeros(3), 0.0, 25)

        return self._success()


def build_robot(env: Any, *, control_mode: str, robot_uid: str) -> GeneratedRobot:
    return GeneratedRobot(env, control_mode=control_mode, robot_uid=robot_uid)
