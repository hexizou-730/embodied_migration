from typing import Any
import numpy as np

from maniskill_backend.dynamic_adapter import ManiSkillDynamicRobot


class GeneratedRobot(ManiSkillDynamicRobot):
    def __init__(self, env: Any, *, control_mode: str, robot_uid: str) -> None:
        super().__init__(
            env,
            control_mode=control_mode,
            robot_uid=robot_uid,
            gripper_open=0.81,
            gripper_close=0.0,
            move_steps=30,
            settle_steps=12,
            max_delta_m=0.04,
        )

    # ------------------------------------------------------------------
    # helpers
    # ------------------------------------------------------------------
    def _safe_move(self, target, gripper, steps):
        target = np.asarray(target, dtype=np.float64).reshape(3)
        self._move_towards(target, gripper=gripper, steps=steps)
        return not self._early_stop()

    def _safe_repeat(self, delta, gripper, steps):
        delta = np.asarray(delta, dtype=np.float64).reshape(3)
        self._repeat_action(delta, gripper=gripper, steps=steps)
        return not self._early_stop()

    def _above(self, pos, height):
        pos = np.asarray(pos, dtype=np.float64).reshape(3)
        return np.array([pos[0], pos[1], pos[2] + height], dtype=np.float64)

    def _entity_moved(self, name, initial, tol=0.005):
        cur = np.asarray(self._entity_pos(name), dtype=np.float64).reshape(3)
        return float(np.linalg.norm(cur - initial)) > tol

    def _tcp_xy_err(self, target_xy):
        tcp = np.asarray(self._tcp_pos(), dtype=np.float64).reshape(3)
        return float(np.linalg.norm(tcp[:2] - np.asarray(target_xy, dtype=np.float64).reshape(2)))

    # ------------------------------------------------------------------
    # policy
    # ------------------------------------------------------------------
    def run_source_actions(self) -> bool:
        half = 0.02

        cubeA = np.asarray(self._entity_pos("cubeA"), dtype=np.float64).reshape(3)
        cubeB = np.asarray(self._entity_pos("cubeB"), dtype=np.float64).reshape(3)

        # ---- 1) Hover above cube A ----
        if not self._safe_move(self._above(cubeA, 0.12), self.gripper_open, 40):
            return False

        # ---- 2) Descend to just above grasp height ----
        pre_grasp = np.array([cubeA[0], cubeA[1], cubeA[2] + half + 0.015],
                             dtype=np.float64)
        if not self._safe_move(pre_grasp, self.gripper_open, 40):
            return False

        # ---- 3) Close gripper at grasp height ----
        grasp_target = np.array([cubeA[0], cubeA[1], cubeA[2] + half],
                                dtype=np.float64)
        if not self._safe_move(grasp_target, self.gripper_close, 40):
            return False

        # ---- 4) Verify grasp by lifting a bit and checking cube A moved ----
        lift_target = self._above(cubeA, 0.10)
        if not self._safe_move(lift_target, self.gripper_close, 40):
            return False

        if not self._entity_moved("cubeA", cubeA, tol=0.01):
            # Grasp failed; retry once with a slightly deeper approach.
            if not self._safe_move(self._above(cubeA, 0.12), self.gripper_open, 40):
                return False
            deeper = np.array([cubeA[0], cubeA[1], cubeA[2] + half - 0.005],
                              dtype=np.float64)
            if not self._safe_move(deeper, self.gripper_open, 40):
                return False
            if not self._safe_move(deeper, self.gripper_close, 40):
                return False
            if not self._safe_move(self._above(cubeA, 0.10), self.gripper_close, 40):
                return False
            if not self._entity_moved("cubeA", cubeA, tol=0.01):
                return False

        # Re-measure cube A after lift (it may have shifted).
        cubeA = np.asarray(self._entity_pos("cubeA"), dtype=np.float64).reshape(3)
        cubeB = np.asarray(self._entity_pos("cubeB"), dtype=np.float64).reshape(3)

        # ---- 5) Move above cube B while carrying A ----
        if not self._safe_move(self._above(cubeB, 0.16), self.gripper_close, 60):
            return False

        # ---- 6) Descend to place cube A on top of cube B ----
        place_z = cubeB[2] + half * 2 + half  # top of B + half of A
        place_target = np.array([cubeB[0], cubeB[1], place_z], dtype=np.float64)
        if not self._safe_move(place_target, self.gripper_close, 40):
            return False

        # ---- 7) Release cube A ----
        if not self._safe_move(place_target, self.gripper_open, 40):
            return False

        # ---- 8) Retreat upward to let cube A settle ----
        if not self._safe_move(self._above(cubeB, 0.16), self.gripper_open, 40):
            return False

        # ---- 9) Settle wait ----
        for _ in range(4):
            if self._early_stop():
                break
            self._repeat_action(np.zeros(3), gripper=self.gripper_open, steps=12)

        # ---- 10) Final verification via official evaluation ----
        try:
            ev = self._official_evaluation()
            if isinstance(ev, dict):
                return bool(ev.get("success", False))
        except Exception:
            pass
        return False


def build_robot(env: Any, *, control_mode: str, robot_uid: str) -> GeneratedRobot:
    return GeneratedRobot(env, control_mode=control_mode, robot_uid=robot_uid)
