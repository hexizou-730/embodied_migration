def _stage_align(self) -> bool:
    cubeB = self._entity_pos("cubeB")
    cubeA = self._entity_pos("cubeA")
    tcp = self._tcp_pos()
    offset = cubeA - tcp
    half = 0.02
    target_z = float(cubeB[2]) + 2.0 * half + float(offset[2])
    align_target = np.array(
        [cubeB[0] - offset[0], cubeB[1] - offset[1], target_z],
        dtype=np.float32,
    )
    for _ in range(14):
        if self._early_stop():
            return False
        cubeA = self._entity_pos("cubeA")
        cubeB = self._entity_pos("cubeB")
        tcp = self._tcp_pos()
        offset = cubeA - tcp
        target_z = float(cubeB[2]) + 2.0 * half + float(offset[2])
        align_target = np.array(
            [cubeB[0] - offset[0], cubeB[1] - offset[1], target_z],
            dtype=np.float32,
        )
        self._move_towards(align_target, gripper=self.gripper_close, steps=5)
        cubeA = self._entity_pos("cubeA")
        cubeB = self._entity_pos("cubeB")
        xy_err = float(np.linalg.norm(cubeA[:2] - cubeB[:2]))
        z_err = abs(float(cubeA[2] - cubeB[2]) - 2.0 * half)
        if xy_err <= 0.006 and z_err <= 0.006:
            break
    cubeA = self._entity_pos("cubeA")
    cubeB = self._entity_pos("cubeB")
    xy_ok = float(np.linalg.norm(cubeA[:2] - cubeB[:2])) <= 0.010
    z_ok = abs(float(cubeA[2] - cubeB[2]) - 2.0 * half) <= 0.010
    grasped = bool(self._is_grasping_entity("cubeA"))
    return bool(xy_ok and z_ok and grasped)
