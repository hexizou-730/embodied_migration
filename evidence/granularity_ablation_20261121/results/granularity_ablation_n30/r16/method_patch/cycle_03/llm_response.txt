def _stage_align(self) -> bool:
    half = 0.02
    # Re-read measured geometry; cubeB may have shifted during lift.
    cubeB = np.array(self._entity_pos("cubeB"), dtype=np.float32)
    cubeA = np.array(self._entity_pos("cubeA"), dtype=np.float32)
    tcp = np.array(self._tcp_pos(), dtype=np.float32)

    target_xy = np.array([cubeB[0], cubeB[1]], dtype=np.float32)
    target_z = float(cubeB[2]) + 2.0 * half

    # Phase 1: raise to a safe hover above the stack target, tracking held-cube offset.
    hover_z = target_z + 0.05
    for _ in range(12):
        if self._early_stop():
            return False
        cubeA = np.array(self._entity_pos("cubeA"), dtype=np.float32)
        tcp = np.array(self._tcp_pos(), dtype=np.float32)
        offset = cubeA - tcp
        desired_cube = np.array([target_xy[0], target_xy[1], hover_z], dtype=np.float32)
        desired_tcp = desired_cube - offset
        delta = desired_tcp - tcp
        if np.linalg.norm(delta) <= 0.008:
            break
        self._move_towards(desired_tcp, gripper=self.gripper_close, steps=self.move_steps)

    # Phase 2: align XY precisely over cubeB while holding hover height.
    for _ in range(12):
        if self._early_stop():
            return False
        cubeB = np.array(self._entity_pos("cubeB"), dtype=np.float32)
        cubeA = np.array(self._entity_pos("cubeA"), dtype=np.float32)
        tcp = np.array(self._tcp_pos(), dtype=np.float32)
        offset = cubeA - tcp
        target_xy = np.array([cubeB[0], cubeB[1]], dtype=np.float32)
        desired_cube = np.array([target_xy[0], target_xy[1], hover_z], dtype=np.float32)
        desired_tcp = desired_cube - offset
        delta = desired_tcp - tcp
        if np.linalg.norm(delta[:2]) <= 0.003:
            break
        self._move_towards(desired_tcp, gripper=self.gripper_close, steps=self.move_steps)

    # Phase 3: descend slowly to final stack height, re-reading cube pose each step.
    for _ in range(14):
        if self._early_stop():
            return False
        cubeB = np.array(self._entity_pos("cubeB"), dtype=np.float32)
        cubeA = np.array(self._entity_pos("cubeA"), dtype=np.float32)
        tcp = np.array(self._tcp_pos(), dtype=np.float32)
        offset = cubeA - tcp
        target_xy = np.array([cubeB[0], cubeB[1]], dtype=np.float32)
        target_z = float(cubeB[2]) + 2.0 * half
        desired_cube = np.array([target_xy[0], target_xy[1], target_z], dtype=np.float32)
        desired_tcp = desired_cube - offset
        delta = desired_tcp - tcp
        if np.linalg.norm(delta) <= 0.005:
            break
        self._move_towards(desired_tcp, gripper=self.gripper_close, steps=self.move_steps)

    # Settle in place so the held cube stabilizes before release.
    for _ in range(4):
        if self._early_stop():
            return False
        self._repeat_action(np.zeros(3), gripper=self.gripper_close, steps=self.settle_steps)

    # Measure completion against the official object-relative predicate.
    cubeA = np.array(self._entity_pos("cubeA"), dtype=np.float32)
    cubeB = np.array(self._entity_pos("cubeB"), dtype=np.float32)
    xy_err = float(np.linalg.norm(cubeA[:2] - cubeB[:2]))
    z_err = float(abs((float(cubeA[2]) - float(cubeB[2])) - 2.0 * half))
    xy_ok = xy_err <= 0.006
    z_ok = z_err <= 0.006
    return bool(xy_ok and z_ok and self._is_grasping_entity("cubeA"))
