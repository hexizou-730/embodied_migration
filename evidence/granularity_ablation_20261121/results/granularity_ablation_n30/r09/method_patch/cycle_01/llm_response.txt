def _stage_align(self) -> bool:
    cubeB = self._entity_pos("cubeB")
    cubeA = self._entity_pos("cubeA")
    # Target cubeA center: directly above cubeB at one full cube height (2 * half_size).
    half = 0.02
    target_z = float(cubeB[2]) + 2.0 * half
    target_xy = np.array([cubeB[0], cubeB[1]], dtype=np.float32)

    # First raise the held cube to a safe hover height above the stack target.
    hover_z = target_z + 0.06
    for _ in range(6):
        if self._early_stop():
            return False
        cubeA = self._entity_pos("cubeA")
        tcp = self._tcp_pos()
        # Command TCP so that the held cube moves toward the hover point.
        offset = cubeA - tcp
        hover = np.array([target_xy[0], target_xy[1], hover_z], dtype=np.float32)
        desired_cube = hover
        cmd = desired_cube - offset
        self._move_towards(cmd, gripper=self.gripper_close, steps=self.move_steps)
        cubeA = self._entity_pos("cubeA")
        if np.linalg.norm(cubeA[:2] - target_xy) <= 0.015 and abs(float(cubeA[2]) - hover_z) <= 0.02:
            break

    # Descend so the held cube center sits at the stack height above cubeB.
    for _ in range(8):
        if self._early_stop():
            return False
        cubeA = self._entity_pos("cubeA")
        tcp = self._tcp_pos()
        offset = cubeA - tcp
        desired_cube = np.array([target_xy[0], target_xy[1], target_z], dtype=np.float32)
        cmd = desired_cube - offset
        self._move_towards(cmd, gripper=self.gripper_close, steps=self.move_steps)
        cubeA = self._entity_pos("cubeA")
        xy_err = float(np.linalg.norm(cubeA[:2] - target_xy))
        z_err = abs(float(cubeA[2]) - target_z)
        if xy_err <= 0.008 and z_err <= 0.008:
            break

    cubeA = self._entity_pos("cubeA")
    xy_ok = bool(np.linalg.norm(cubeA[:2] - target_xy) <= 0.012)
    z_ok = bool(abs(float(cubeA[2]) - target_z) <= 0.012)
    return bool(xy_ok and z_ok and self._is_grasping_entity("cubeA"))
